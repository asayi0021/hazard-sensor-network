51/// List of read/writable registers (and their address) on the gas sensor.
52/// For more details, see [BME680 datasheet][registers].
53///
54/// [registers]: https://www.bosch-sensortec.com/media/boschsensortec/downloads/datasheets/bst-bme680-ds001.pdf
55#[derive(Format, Clone)]
56pub enum Registers {
57    Status = 0x73,
58    Reset = 0xE0,
59    Id = 0xD0,
60    Config = 0x75,
61    CtrlMeas = 0x74,
62    CtrlHum = 0x72,
63    CtrlGas1 = 0x71,
64    CtrlGas0 = 0x70,
65
66    // Gas control registers
67    // There are actually 10 registers for each of the 10 set-points.
68    // The recorded address is just the addressof set-point 0. The next 9 address in sequence correspond to
69    // each subsequent set-point sequentially.
70    // x is the set-point (ranging from 0 to 9).
71    // To get to a certain X value, just add X to the base address.
72    GasWaitX = 0x64,
73    ResHeatX = 0x5A,
74    IdacHeatX = 0x50,
75
76    GasRLsb = 0x2B,
77    GasRMsb = 0x2A,
78    HumLsb = 0x26,
79    HumMsb = 0x25,
80    TempXlsb = 0x24,
81    TempLsb = 0x23,
82    TempMsb = 0x22,
83    PresXlsb = 0x21,
84    PressLsb = 0x20,
85    PressMsb = 0x1F,
86    MeasStatus0 = 0x1D,
87    Calib1 = 0x00,
88    Calib2 = 0xEB,
89}

149    /// Write to one or more registers
150    async fn write(&mut self, reg_addr: Registers, data: &[u8]) -> Result<(), SensorError> {
151        let reg_cp = reg_addr.clone();
152
153        // Build one contiguous buffer: [register address, data...].
154        let mut buf = [0u8; 2];
155        buf[0] = reg_addr as u8;
156        buf[1..1 + data.len()].copy_from_slice(data);
157
158        match self.i2c.write(
159            self.addr,
160            &buf[..1 + data.len()],
161        ).await {
162            Ok(_) => {
163                trace!("write to register [{:?}]: {:#010b}", reg_cp, data);
164                Ok(())
165            }
166            Err(e) => {
167                let err: SensorError = e.into();
168                error!("Could not write register [{:?}]: {:?}", reg_cp, err);
169                Err(err)
170            }
171        }
172    }

174    /// Read from a single register
175    async fn read(&mut self, reg_addr: Registers) -> Result<u8, SensorError> {
176        let mut read_buf = [0];
177        let reg_cp = reg_addr.clone();
178        match self
179            .i2c
180            .write_read(self.addr, &[reg_addr as u8], &mut read_buf).await
181        {
182            Ok(_) => {
183                trace!("read from register [{:?}]: {:#010b}", reg_cp, read_buf);
184                Ok(read_buf[0])
185            }
186            Err(e) => {
187                let err: SensorError = e.into();
188                error!("Could not read register [{:?}]: {:?}", reg_cp, err);
189                Err(err)
190            }
191        }
192    }

211    /// Reads the gas-heater calibration/trim data from the sensor.
212    ///
213    /// Must be called once (e.g. from an `init()` step) before computing any
214    /// res_heat_x value. Must pass in raw I2C bus and address since it is called
215    /// before constructor is fully initialised.
216    async fn read_calibration_raw(i2c: &mut I2C, addr: u8) -> Result<CalibrationData, SensorError> {
217        // res_heat_val (0x00) and res_heat_range (bits 5:4 of 0x02)
218        let mut buf0 = [0u8; 4];
219        i2c.write_read(addr, &[0x00], &mut buf0).await?;
220        let res_heat_val = buf0[0] as i8;
221        let res_heat_range = (buf0[2] & 0x30) >> 4;
222        // dividing by 16 allows for sign extension
223        let range_switching_error = ((buf0[3] & 0xF0) as i8) / 16;
224
225        // Coefficient block 1: registers 0x89..=0xA1 (25 bytes)
226        let mut c1 = [0u8; 25];
227        i2c.write_read(addr, &[0x89], &mut c1).await?;
228
229        // Coefficient block 2: registers 0xE1..=0xF0 (16 bytes)
230        let mut c2 = [0u8; 16];
231        i2c.write_read(addr, &[0xE1], &mut c2).await?;
232
233        // --- Temperature ---
234        let par_t1 = i16::from_le_bytes([c2[8], c2[9]]); // 0xE9, 0xEA
235        let par_t2 = i16::from_le_bytes([c1[1], c1[2]]); // 0x8A, 0x8B
236        let par_t3 = c1[3] as i8; // 0x8C
237
238        // --- Pressure ---
239        let par_p1 = u16::from_le_bytes([c1[5], c1[6]]); // 0x8E, 0x8F
240        let par_p2 = i16::from_le_bytes([c1[7], c1[8]]); // 0x90, 0x91
241        let par_p3 = c1[9] as i8; // 0x92
242        let par_p4 = i16::from_le_bytes([c1[11], c1[12]]); // 0x94, 0x95
243        let par_p5 = i16::from_le_bytes([c1[13], c1[14]]); // 0x96, 0x97
244        let par_p7 = c1[15] as i8; // 0x99
245        let par_p6 = c1[16] as i8; // 0x98
246        let par_p8 = i16::from_le_bytes([c1[19], c1[20]]); // 0x9C, 0x9D
247        let par_p9 = i16::from_le_bytes([c1[21], c1[22]]); // 0x9E, 0x9F
248        let par_p10 = c1[23] as i8; // 0xA0
249
250        // --- Humidity (par_h1/par_h2 are 12-bit values sharing one byte, nibble-packed) ---
251        let par_h1 = (((c2[2] as i16) << 4) | ((c2[1] as i16) & 0x0F)) as u16; // 0xE3 MSB, 0xE2 lo-nibble
252        let par_h2 = (((c2[0] as i16) << 4) | ((c2[1] as i16) >> 4)) as i16; // 0xE1 MSB, 0xE2 hi-nibble
253        let par_h3 = c2[3] as i8; // 0xE4
254        let par_h4 = c2[4] as i8; // 0xE5
255        let par_h5 = c2[5] as i8; // 0xE6
256        let par_h6 = c2[6] as u8; // 0xE7
257        let par_h7 = c2[7] as i8; // 0xE8
258
259        // --- Gas heater ---
260        let par_g2 = i16::from_le_bytes([c2[10], c2[11]]); // 0xEB, 0xEC
261        let par_g1 = c2[12] as i8; // 0xED
262        let par_g3 = c2[13] as i8; // 0xEE
263
264        trace!(
265            "calibration: t1={:?} t2={:?} t3={:?} p1={:?} h1={:?} h2={:?} g1={:?} g2={:?} g3={:?}",
266            par_t1, par_t2, par_t3, par_p1, par_h1, par_h2, par_g1, par_g2, par_g3
267        );
268
269        Ok(CalibrationData {
270            par_t1,
271            par_t2,
272            par_t3,
273            par_p1,
274            par_p2,
275            par_p3,
276            par_p4,
277            par_p5,
278            par_p6,
279            par_p7,
280            par_p8,
281            par_p9,
282            par_p10,
283            par_h1,
284            par_h2,
285            par_h3,
286            par_h4,
287            par_h5,
288            par_h6,
289            par_h7,
290            par_g1,
291            par_g2,
292            par_g3,
293            res_heat_range,
294            res_heat_val,
295            range_switching_error,
296        })
297    }

299    /// Computes the res_heat_x register value using integer-only arithmetic,
300    /// matching Bosch's official fixed-point reference implementation.
301    async fn calc_res_heat(&mut self, target_temp_c: i32, amb_temp_c: i32) -> u8 {
302        let var1 = ((amb_temp_c * self.calibration.par_g3 as i32) / 10) << 8;
303        let var2 = (self.calibration.par_g1 as i32 + 784)
304            * (((((self.calibration.par_g2 as i32 + 154009) * target_temp_c * 5) / 100) + 3276800)
305                / 10);
306        let var3 = var1 + (var2 >> 1);
307        let var4 = var3 / (self.calibration.res_heat_range as i32 + 4);
308        let var5 = (131 * self.calibration.res_heat_val as i32) + 65536;
309        let res_heat_x100 = ((var4 / var5) - 250) * 34;
310        let res_heat_x = (res_heat_x100 + 50) / 100;
311        res_heat_x as u8
312    }

314    /// Encodes a heating duration in milliseconds into the gas_wait_x register format
315    /// bits<5:0> = value (0-63), bits<7:6> = multiplier factor (00=x1, 01=x4, 10=x16, 11=x64).
316    ///
317    /// Max representable duration is 63 * 64 = 4032ms.
318    async fn encode_gas_wait(duration_ms: u16) -> u8 {
319        const MAX_DURATION_MS: u16 = 0xFC0; // 4032ms
320
321        if duration_ms >= MAX_DURATION_MS {
322            0xFF // saturate at the maximum representable duration
323        } else {
324            let mut dur = duration_ms;
325            let mut factor: u8 = 0;
326
327            while dur > 0x3F {
328                dur /= 4;
329                factor += 1;
330            }
331
332            dur as u8 + (factor * 64)
333        }
334    }

336    /// Initial configuration based on BME860 quick start guide.
337    ///
338    /// Needs to be called after `new()`.
339    pub async fn init_config(&mut self) -> Result<(), SensorError> {
340        debug!("Configuring GAS SENSOR...");
341        // Set humidity oversampling
342        let ctrl_hum = self.read(Registers::CtrlHum).await?;
343        self.write(Registers::CtrlHum, &[(ctrl_hum & !0b111) | 0b001]).await?;
344
345        // Set temperature oversampling (osrs_t=2), pressure oversampling (osrs_p=5), mode=sleep
346        self.write(Registers::CtrlMeas, &[0b0101_0100]).await?;
347
348        let mut read_buf = [0; 3];
349        self.read_bytes(Registers::CtrlHum, &mut read_buf).await?;
350        trace!("oversampling registers: {:#010b}", read_buf);
351
352        // Set hot plate temperature set-point to X and heating duration to 100ms.
353        let duration = Self::encode_gas_wait(150).await;
354        // 320C seems to be typical Bosch application target temperature, give it 25 degrees for default ambient temperature for set up.
355        let res_heat = self.calc_res_heat(320, 25).await;
356        self.write(Registers::GasWaitX, &[duration]).await?;
357        self.write(Registers::ResHeatX, &[res_heat]).await?;
358        // Configure sensor to use set-point 0 and enable gas measurement. (Writes to nv_conv<3:0> and run_gas_l)
359        let ctrl_gas_1 = self.read(Registers::CtrlGas1).await?;
360        self.write(Registers::CtrlGas1, &[(ctrl_gas_1 & !0b1_1111) | 0b1_0000]).await?;
361
362        Ok(())
363    }

367        let ctrl_meas = self.read(Registers::CtrlMeas).await?;

380            Err(e) => {
381                error!("Could not set gas sensor mode: {:?}", e);
382                Err(e)
383            }
384        }

387    /// Computes gas resistance (in Ohms) from raw ADC gas reading and gas_range.
388    async fn calc_gas_resistance(&self, gas_adc: u16, gas_range: u8) -> i32 {
389        let range_switching_error = self.calibration.range_switching_error as i64;
390        let gas_range = gas_range as usize;
391
392        let var1: i64 = ((1340 + (5 * range_switching_error))
393            * (LOOKUP_TABLE_1[gas_range] as i64))
394            >> 16;
395
396        let var2: i64 = ((gas_adc as i64) << 15) - (1i64 << 24) + var1;
397
398        let gas_res: i32 = ((((LOOKUP_TABLE_2[gas_range] as i64 * var1) >> 9) + (var2 >> 1)) / var2) as i32;
399
400        gas_res
401    }

403    /// Computes compensated gas sensor resitance output data in Ohms.
404    // Seems like IAQ is only provided through the BSEC software
405    async fn get_gas_resistance(&mut self) -> Result<i32, SensorError> {
406        // Covers MSB (0x2A) and LSB (0x2B) registers for gas quality
407        let mut read_buf = [0; 2];
408        self.read_bytes(Registers::GasRMsb, &mut read_buf).await?;
409        let gas_adc: u16 = ((read_buf[0] as u16) << 2)
410            | ((read_buf[1] & 0b1100_0000) as u16 >> 6);
411        let gas_range = read_buf[1] & 0b0000_1111;
412        let final_res = self.calc_gas_resistance(gas_adc, gas_range).await;
413        debug!("Gas resistance read success: {} Ohms", final_res);
414        Ok(final_res)
415    }

417    /// Computes compensated temperature (in degrees Celsisus) from raw ADC temperature.
418    ///
419    /// Also returns t_fine that is used for calculating the compensated pressure.
420    async fn calc_temp(&mut self, temp_adc: u32) -> [i32; 2] {
421        let var1 = (temp_adc as i32 >> 3) - ((self.calibration.par_t1 as i32) << 1);
422        let var2 = (var1 * self.calibration.par_t2 as i32) >> 11;
423        let var3 =
424            ((((var1 >> 1) * (var1 >> 1)) >> 12) * ((self.calibration.par_t3 as i32) << 4)) >> 14;
425        let t_fine = var2 + var3;
426        let temp_comp = ((t_fine * 5) + 128) >> 8;
427        [temp_comp, t_fine]
428    }

430    /// Returns compensated temperature in degrees Celsius.
431    async fn get_temp(&mut self) -> Result<[i32; 2], SensorError> {
432        // Covers MSB (0x22), LSB (0x23), XLSB (0x24) registers for temperature
433        let mut read_buf = [0; 3];
434        self.read_bytes(Registers::TempMsb, &mut read_buf).await?;
435        let temp_adc: u32 = ((read_buf[0] as u32) << 12)
436            | ((read_buf[1] as u32) << 4)
437            | ((read_buf[2] as u32) >> 4);
438        let final_temp = self.calc_temp(temp_adc).await;
439        debug!("Temperature read success: {} centidegrees Celsius", final_temp[0]);
440        Ok(final_temp)
441    }

443    /// Computes compensated humidity (in milli%) from raw ADC humidity and
444    /// temp_comp (the calculated temperature compensation value).
445    async fn calc_humidity(&self, hum_adc: u16, temp_comp: i32) -> u32 {
446        let par_h1 = self.calibration.par_h1 as i32;
447        let par_h2 = self.calibration.par_h2 as i32;
448        let par_h3 = self.calibration.par_h3 as i32;
449        let par_h4 = self.calibration.par_h4 as i32;
450        let par_h5 = self.calibration.par_h5 as i32;
451        let par_h6 = self.calibration.par_h6 as i32;
452        let par_h7 = self.calibration.par_h7 as i32;
453
454        let var1 = (hum_adc as i32) - (par_h1 << 4) - (((temp_comp * par_h3) / 100) >> 1);
455
456        let var2 = (par_h2
457            * (((temp_comp * par_h4) / 100)
458                + (((temp_comp * ((temp_comp * par_h5) / 100)) >> 6) / 100)
459                + (1 << 14)))
460            >> 10;
461
462        let var3 = var1 * var2;
463
464        let var4 = ((par_h6 << 7) + ((temp_comp * par_h7) / 100)) >> 4;
465
466        let var5 = ((var3 >> 14) * (var3 >> 14)) >> 10;
467
468        let var6 = (var4 * var5) >> 1;
469
470        let hum_comp = (((var3 + var6) >> 10) * 1000) >> 12;
471
472        hum_comp as u32
473    }

487    /// Computes compensated pressure (in Pa) from raw ADC pressure and t_fine
488    /// (the intermediate value produced during temperature compensation).
489    // AI Note: Rust panics on integer overflow in debug builds (unlike C, which silently wraps). This formula is designed by Bosch to stay within i32 range for realistic sensor inputs, so it should be fine in practice — but if you ever do hit a panic here during testing, it's a legitimate signal that an upstream value (bad calibration read, corrupted ADC data) is out of the expected range, not something to just wrap around and ignore.
490    async fn calc_pressure(&self, press_adc: u32, t_fine: i32) -> u32 {
491        let par_p1 = self.calibration.par_p1 as i32;
492        let par_p2 = self.calibration.par_p2 as i32;
493        let par_p3 = self.calibration.par_p3 as i32;
494        let par_p4 = self.calibration.par_p4 as i32;
495        let par_p5 = self.calibration.par_p5 as i32;
496        let par_p6 = self.calibration.par_p6 as i32;
497        let par_p7 = self.calibration.par_p7 as i32;
498        let par_p8 = self.calibration.par_p8 as i32;
499        let par_p9 = self.calibration.par_p9 as i32;
500        let par_p10 = self.calibration.par_p10 as i32;
501
502        let mut var1: i32 = (t_fine >> 1) - 64000;
503
504        let mut var2: i32 = ((((var1 >> 2) * (var1 >> 2)) >> 11) * par_p6) >> 2;
505        var2 += (var1 * par_p5) << 1;
506        var2 = (var2 >> 2) + (par_p4 << 16);
507
508        var1 =
509            (((((var1 >> 2) * (var1 >> 2)) >> 13) * (par_p3 << 5)) >> 3) + ((par_p2 * var1) >> 1);
510        var1 >>= 18;
511        var1 = ((32768 + var1) * par_p1) >> 15;
512
513        let mut press_comp: u32 = (1_048_576 - press_adc as i32 - (var2 >> 12)) as u32 * 3125;
514
515        press_comp = if press_comp >= (1 << 30) {
516            (press_comp / var1 as u32) << 1
517        } else {
518            (press_comp << 1) / var1 as u32
519        };
520
521        let var1_final = (par_p9 * ((((press_comp >> 3) * (press_comp >> 3)) >> 13) as i32)) >> 12;
522        let var2_final = ((press_comp >> 2) as i32 * par_p8) >> 13;
523        let var3_final = ((press_comp >> 8) as i32
524            * (press_comp >> 8) as i32
525            * (press_comp >> 8) as i32
526            * par_p10)
527            >> 17;
528
529        press_comp = ((press_comp as i32)
530            + ((var1_final + var2_final + var3_final + (par_p7 << 7)) >> 4))
531            as u32;
532
533        press_comp
534    }
535
536    /// Returns compensated pressure in Pa.
537    async fn get_pressure(&mut self, t_fine: i32) -> Result<u32, SensorError> {
538        // Covers MSB (0x1F), LSB (0x20) and XLSB (0x21) registers for humidity
539        let mut read_buf = [0; 3];
540        self.read_bytes(Registers::PressMsb, &mut read_buf).await?;
541        let press_adc: u32 = ((read_buf[0] as u32) << 12)
542            | ((read_buf[1] as u32) << 4)
543            | ((read_buf[2] as u32) >> 4);
544        let final_press = self.calc_pressure(press_adc, t_fine).await;
545        debug!("Pressure read success: {} Pa", final_press);
546        Ok(final_press)
547    }

549    /// Check if measured data are ready to be read.
550    async fn check_if_ready(&mut self) -> Result<(), SensorError> {
551        loop {
552            let meas_status = self.read(Registers::MeasStatus0).await?;
553            if (meas_status & 0b1000_0000) != 0 {
554                return Ok(());
555            }
556            Timer::after_millis(100).await;
557        }
558    }

560    /// Get all four measurement readings and convert them into a more readable unit.
561    /// Air quality index is currently an arbitrary unit ranging between 0-2.
562    /// 0 = Clean air, 1 = Polluted air, 2 = Heavily polluted air (sensor is saturated)
563    ///
564    /// The thresholds will need to be retuned after proper testing.
565    pub async fn get_measurements(&mut self) -> Result<(f64, f64, f64, u8), SensorError> {
566        debug!("Starting GAS SENSOR measurement...");
567        self.set_mode(GSMode::Forced).await?;
568        self.check_if_ready().await?;
569
570        let temp = self.get_temp().await?;
571        // convert from centidegrees Celsius to Celsius
572        let temperature = temp[0] as f64 / 100.0;
573        // convert from milli% to %
574        let humidity = self.get_humidity(temp[0]).await? as f64 / 1000.0;
575        // convert from Pa to hPa
576        let pressure = self.get_pressure(temp[1]).await? as f64 / 100.0;
577        let gas_res = self.get_gas_resistance().await?;
578
579        debug!("GAS SENSOR measurements done!");
580
581        // Convert resistance value of gas measurement to an air quality index, based
582        // on the range of the resistance value. Thresholds are currently set through
583        // very basic testing.
584        let mut air_quality = 0;
585        if gas_res >= 35000 {
586            air_quality = 0;    // clean
587        } else if (5000..35000).contains(&gas_res) {
588            air_quality = 1;    // polluted
589        } else if gas_res < 5000 {
590            air_quality = 2;    // heavily polluted
591        }
592
593        info!("Temperature: {} degrees Celsius, Humidity: {} r.H%, Pressure: {} hPa, Air Quality: {}", temperature, humidity, pressure, gas_res);
594
595        debug!("GAS SENSOR going to sleep...");
596        self.set_mode(GSMode::Sleep).await?;
597
598        Ok((temperature, humidity, pressure, air_quality))
599    }
