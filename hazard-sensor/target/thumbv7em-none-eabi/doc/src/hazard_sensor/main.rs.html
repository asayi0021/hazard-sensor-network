78impl NRF52840 {
79    /// Initialise a new nRF52840 struct
80    pub fn new(
81        twispi0: Peri<'static, peripherals::TWISPI0>,
82        sda: Peri<'static, peripherals::P0_13>,
83        scl: Peri<'static, peripherals::P0_14>,
84        saadc_pin: Peri<'static, peripherals::SAADC>,
85        adc_battery_pin: Peri<'static, peripherals::P0_05>,
86    ) -> Self {
87        // Initialise config for each bus
88        let i2c_config = twim::Config::default();
89        let adc_config = saadc::Config::default();
90
91        // Initialise saadc channels
92        let bat_channel = ChannelConfig::single_ended(adc_battery_pin);
93
94        // Initialize the i2c drivers
95        let i2c = twim::Twim::new(twispi0, Irqs, sda, scl, i2c_config, TX_BUFF1.take());
96
97        // Intitialise saadc
98        let saadc = Saadc::new(saadc_pin, Irqs, adc_config, [bat_channel]);
99
100        NRF52840 {
101            i2c,
102            saadc,
103        }
104    }
105}

107/// Sx1262 Object for initalising the transceiver and it's assigned parameters.
108pub struct SX1262 {
109    lora_radio: LoRa<Sx126x<ExclusiveDevice<spim::Spim<'static>, gpio::Output<'static>, embassy_time::Delay>,
110            iv::GenericSx126xInterfaceVariant<gpio::Output<'static>, gpio::Input<'static>>,
111            Sx1262,>,embassy_time::Delay,>,
112    mod_params: ModulationParams,
113    rx_pkt_params: PacketParams,
114    tx_pkt_params: PacketParams,
115}
116
117impl SX1262 {
118    /// Initialise a new SX1262 struct
119    pub async fn new(
120        // SX1262 pins
121        reset: Peri<'static, peripherals::P1_06>,    // In RAK4630 datasheet, the DIO2
122        busy: Peri<'static, peripherals::P1_14>,     // is used to control antenna switch,
123        dio1: Peri<'static, peripherals::P1_15>,     // and that the GPIO 1.07 pin should
124        rf_tx_en: Peri<'static, peripherals::P1_07>, // and that the GPIO 1.07 pin should
125        rf_rx_en: Peri<'static, peripherals::P1_05>, // not be initialised.
126        // spi bus pins
127        spi3: Peri<'static, peripherals::SPI3>,
128        sck: Peri<'static, peripherals::P1_11>,
129        miso: Peri<'static, peripherals::P1_13>,
130        mosi: Peri<'static, peripherals::P1_12>,
131        nss: Peri<'static, peripherals::P1_10>,
132    ) -> Self {
133        // Convert pins to outputs/inputs
134        let reset_o = Output::new(reset, Level::High, OutputDrive::Standard);        //NRESET P1.06
135        let busy_i = Input::new(busy, Pull::None);                                                          //BUSY P1.14
136        let dio1_i = Input::new(dio1, Pull::Down);                                                          //DIO1 P1.15
137        let rf_tx_en_o = Output::new(rf_tx_en, Level::High, OutputDrive::Standard);
138        let rf_rx_en_o = Output::new(rf_rx_en, Level::High, OutputDrive::Standard);  //ANT_SW P1.05
139        // nss_pin as output
140        let nss_o = Output::new(nss, Level::High, OutputDrive::Standard);
141
142        // Setup Sx1262 config
143        let sx1262_config = sx126x::Config {
144            chip: Sx1262,
145            tcxo_ctrl: Some(sx126x::TcxoCtrlVoltage::Ctrl1V8),
146            use_dcdc: true,
147            rx_boost: true,
148        };
149
150        // Initialise InterfaceVariant lora-phy object
151        let iv = match lora_phy::iv::GenericSx126xInterfaceVariant::new(reset_o, dio1_i, busy_i, Some(rf_rx_en_o), Some(rf_tx_en_o)) {
152            Ok(iv) => iv,
153            Err(e) => {
154                defmt::error!("failed to build SX1262 interface variant: {:?}", defmt::Debug2Format(&e));
155                panic!("interface variant intialisation failed");
156            }
157        };
158
159        // Setup spi config for Sx1262
160        let spi_config = spim::Config::default();
161
162        // Initialise the SPI driver between the nRF52 and the Sx1262
163        let spi = spim::Spim::new(spi3, Irqs, sck, miso, mosi, spi_config);
164
165        // Initialise spi_device to pass to sx1262 constructor
166        let spi_device = ExclusiveDevice::new(spi, nss_o, Delay)
167            .expect("failed to build spi_device");
168
169        // Initialise Sx1262
170        let sx1262 = Sx126x::new(spi_device, iv, sx1262_config);
171
172        // Initialise LoRa radio for MeshCore LoRa communications
173        let mut lora_radio = LoRa::new(sx1262, false, Delay)
174            .await //maybe need to extract from result here?
175            .expect("failed to initalise SX1262 - lora_radio object");
176
177        // Initialise modulation parameters attached to lora_radio field of SX1262 object
178        let mod_params: ModulationParams = lora_radio.create_modulation_params(SPREADING_FACTOR, BANDWIDTH, CODING_RATE, FREQ_HZ)
179            .expect("failed to create modulation params in SX1262 constructor.");
180
181        // Initialise recieved packet parameters attached to lora_radio field of SX1262 object - these parameters are
182        // assumed as constants since the user that sends a query should have the parameters.
183        let rx_pkt_params: PacketParams = lora_radio.create_rx_packet_params(8, false, 255, true, false, &mod_params)
184            .expect("failed to create rx packet params in SX1262 constructor.");
185
186        // Initialise transmitted packet parameters attached to lora_radio field of SX1262 object.
187        // These parameters are fixed as stated in LHN documentation.
188        let tx_pkt_params: PacketParams = lora_radio.create_tx_packet_params(8, false, true, false, &mod_params)
189            .expect("failed to create tx packet params in SX1262 constructor.");
190
191        // Construct SX1262 object
192        SX1262 {
193            lora_radio,
194            mod_params,
195            rx_pkt_params,
196            tx_pkt_params,
197        }
198    }
199}

205/// Main - initialises transceiver and peripherals, then spawns the radio_task.
206#[embassy_executor::main]
207async fn main(_spawner: Spawner) {
208    info!("Main start");
209
210    // Initialisation of peripherals struct p
211    let p = embassy_nrf::init(Default::default());
212
213    // gpio pins for MIKROE-4416 (watchdog).
214    // IMPORTANT: this MUST NOT be P0_17 (IO1) or P1_02 (IO2) -- those are the
215    // reserved WisBlock pins that toggle the 3V3_S peripheral power rail and
216    // the peripheral-module power enable respectively. Using them here would
217    // mean every send_pulse()/set_window() call also cycles power to every
218    // sensor on the I2C bus. P0_15 (RX1) and P0_16 (TX1) are genuinely free.
219    let wdi_pin = Output::new(p.P0_15, Level::Low, OutputDrive::Standard); // WDI  (RX1)
220    let s1_pin = Output::new(p.P0_16, Level::Low, OutputDrive::Standard); // SET1 (TX1)
221
222    let mut watchdog = WatchdogTimer::new(wdi_pin, s1_pin).expect("Failed to initialise WATCHDOG");
223    watchdog.set_window(WatchdogWindow::Disabled).expect("Failed to disable WATCHDOG");
224    info!("Watchdog disabled");
225
226    // Arm the watchdog in 1:2 ratio mode. With SET0 hardwired to 3V3 and CWD
227    // left floating (NC), this gives tWDL(max) ~= 920 ms and tWDU(min) ~= 1360 ms.
228    watchdog.set_window(WatchdogWindow::Ratio1To2).expect("Failed to set WATCHDOG window");
229    info!("Watchdog set to 1:2 ratio");
230    // Wait out t_WD-setup (150us datasheet minimum) with margin before WDI is recognized.
231    Timer::after_micros(500).await;
232    // Send the mandatory first pulse, well inside tWDU(min) ~= 1360 ms.
233    watchdog.send_pulse().await.expect("Failed to send WATCHDOG pulse");
234    info!("Watchdog first pulse sent");
235
236    // Hand the now-armed watchdog off to a dedicated task that feeds it for
237    // the rest of the program's life. Spawned here, before the (potentially
238    // slower, I2C-bound) sensor init below, so the feed ticker is already
239    // running well within the ~1.36s deadline for the *next* pulse.
240    _spawner.spawn(watchdog_task(watchdog)).expect("Failed to spawn WATCHDOG task");
241
242    // nRF52 pin definitions to pass to constructors
243    // i2c pins
244    let twispi0 = p.TWISPI0;
245    let twispi1 = p.TWISPI1;
246    let sda1 = p.P0_13;
247    let scl1 = p.P0_14;
248    // adc pins
249    let saadc_pin = p.SAADC;
250    let adc_battery_pin = p.P0_05;  // pin is AIN2, WB_A0
251
252    // SX1262 pin definitions to pass to constructor
253    let reset = p.P1_06;
254    let busy = p.P1_14;
255    let dio1 = p.P1_15;
256    let rf_tx_en = p.P1_07;
257    let rf_rx_en = p.P1_05;
258    // spi pins for SX1262
259    let spi3 = p.SPI3;
260    let sck = p.P1_11;
261    let miso = p.P1_13;
262    let mosi = p.P1_12;
263    let nss = p.P1_10;
264
265    // Initialisation of RAK4630 objects
266    let mcu = NRF52840::new(twispi0,sda1,scl1, saadc_pin, adc_battery_pin);
267    mcu.saadc.calibrate().await;
268    let radio = SX1262::new(reset, busy, dio1, rf_tx_en, rf_rx_en, spi3, sck, miso, mosi, nss).await;
269
270    // Initialisation of shared I2C bus
271    let i2c_bus = Mutex::new(mcu.i2c);
272    let i2c_bus = I2C_BUS.init(i2c_bus);
273    // Create seperate objects to pass to constructors for shared I2C bus
274    let gas_i2c = I2cDevice::new(i2c_bus);
275    let rain_i2c = I2cDevice::new(i2c_bus);
276    let adc_i2c = I2cDevice::new(i2c_bus);
277
278
279
280    // ADC sensors init - soil moisture (sms) and wind speed (wss)
281    let mut adc = match AdcSensors::new(mcu.saadc, adc_i2c, ADC_SENSOR_ADDR).await { // currently on same i2c bus as tbs
282        Ok(sensor) => {
283            info!("ADC SENSORS initialised.");
284            sensor
285        },
286        Err(e) => panic!("Could not intialise ADC SENSOR: {:?}", e),
287    };
288    match adc.init_config().await {
289        Ok(_) => info!("ADC SENSORS configuration success."),
290        Err(err) => panic!("Failed to configure ADC SENSORS: {:?}", err),
291    };
292
293    // Gas sensor initialisation
294    let mut aqs = match GasSensor::new(gas_i2c, GAS_SENSOR_ADDR).await { // currently on same i2c bus as tbs
295        Ok(sensor) => {
296            info!("GAS SENSOR initialised.");
297            sensor
298        },
299        Err(e) => panic!("Could not intialise GAS SENSOR: {:?}", e),
300    };
301    match aqs.init_config().await {
302        Ok(_) => info!("GAS SENSOR configuration success."),
303        Err(err) => panic!("Failed to configure GAS SENSOR: {:?}", err),
304    };
305
306    // Tipping bucket (rainfall) sensor initialisation
307    let mut tbs =
308        match RainfallSensor::new(rain_i2c, RAINFALL_SENSOR_ADDR).await {
309            Ok(sensor) => {
310                info!("RAINFALL SENSOR initialised.");
311                sensor
312            },
313            Err(e) => panic!("Could not intialise RAINFALL SENSOR: {:?}", e),
314        };
315    match tbs.init_config().await {
316        Ok(_) => info!("RAINFALL SENSOR configuration success."),
317        Err(err) => panic!("Failed to configure RAINFALL SENSOR: {:?}", err),
318    };
319
320    // Start radio task - sends queued transmissions and handles recieved packets
321    _spawner.spawn(radio_task(radio, aqs, adc, tbs)).unwrap();
322}

329/// radio_task drains the MESHCORE_TX_BUFF when it is not empty, and calls
330/// frame_handler when anything is recieved by the transceiver. Now handles
331/// period broadcasts as well.
332#[embassy_executor::task]
333async fn radio_task(
334    mut radio: SX1262,
335    // mut wds: WindDirectionSensor,
336    mut aqs: GasSensor<I2cShared>,
337    mut adc: AdcSensors<I2cShared>,
338    mut tbs: RainfallSensor<I2cShared>,
339){
340    // Initialise recieved packet buffer locally
341    let mut meshcore_rx_buf = [0u8; MAX_PACKET_LEN];
342
343    let mut wind_history = WindHistory::new();
344
345    // Timers for telemetry frequency management, to change frequency change the constants.
346    let mut wind_ticker = Ticker::every(Duration::from_secs(WIND_SAMPLE_PERIOD_SECS));
347    let mut broadcast_ticker = Ticker::every(Duration::from_secs(BROADCAST_PERIOD));
348
349    loop {
350        // Set transceiver to receive mode to listen for incoming packets
351        radio.lora_radio.prepare_for_rx(RxMode::Continuous, &radio.mod_params, &radio.rx_pkt_params)
352            .await
353            .expect("Prepare for rx (continuous mode) failed");
354
355        // Using Either purposefully sets a race condition. Whichever function call
356        // returns result first is taken. If a packet arrives for RX to process
357        // while a TX is being sent out the recieved packet will be dropped.
358        // This currently poses no issues as the only incoming packets of note
359        // are requests and all the broadcasts send out the same data as a
360        // response. However this should be investigated as a possible breakpoint.
361        match select4(
362            radio.lora_radio.rx(&radio.rx_pkt_params, &mut meshcore_rx_buf),
363            MESHCORE_TX_BUFF.receive(),
364            broadcast_ticker.next(),
365            wind_ticker.next(),
366        ).await {
367            // Possibility 1: Packet recieved (success)
368            Either4::First(Ok((len, status))) => {
369                let frame_data = &meshcore_rx_buf[..len as usize];
370                info!("RX {} bytes, rssi={} snr={}", len, status.rssi, status.snr);
371                frame_handler(frame_data, &mut aqs, &mut adc, &mut tbs, &mut wind_history).await;
372            }
373            // Possibility 1: Packet recieved (failure)
374            Either4::First(Err(e)) => warn!("radio rx error {:?}", defmt::Debug2Format(&e)),
375            // Possibility 2: Packet in buffer ready to send out
376            Either4::Second(frame) => send_frame(&mut radio, &frame).await,
377            // Possibility 3: Periodic broadcast reached
378            Either4::Third(()) => {
379                let r = poll_all(&mut aqs, &mut adc, &mut tbs, &mut wind_history).await;
380                if let Err(e) = network::send_group_text_sensor_data(&r) {
381                    error!("failed to send GRP_TXT broadcast: {:?}", defmt::Debug2Format(&e));
382                } else {
383                    info!("public channel broadcast sent");
384                }
385            }
386            Either4::Fourth(()) => {
387                match adc.get_wind_speed().await {
388                    Ok(kmh) => wind_history.push(kmh),
389                    Err(e) => warn!("wind sample failed, skipping: {:?}", defmt::Debug2Format(&e)),
390                }
391            }
392        }
393    }
394}

401/// Feeds the watchdog for the remainder of the program's life, once it has
402/// already been armed and given its mandatory first pulse in `main`.
403///
404/// With SET0 hardwired to 3V3, SET1 driven by this GPIO, and CWD left
405/// floating (NC), the 1:2 ratio window is tWDL(max) ~= 920 ms and
406/// tWDU(min) ~= 1360 ms (typ. 800 ms / 1.6 s). 1100 ms sits with margin on
407/// both sides of that window -- ~180 ms above the lower boundary and
408/// ~260 ms below the upper boundary -- to absorb scheduler jitter and
409/// timing tolerance. A `Ticker` is used rather than repeated `Timer::after`
410/// calls so the period doesn't drift by however long `send_pulse` itself
411/// takes to run.
412///
413/// This task is not expected to return; if `send_pulse` ever errors, that's
414/// logged but pulsing continues, since giving up here would guarantee a
415/// watchdog-triggered reset rather than risk one.
416#[embassy_executor::task]
417async fn watchdog_task(mut watchdog: WatchdogTimer<Output<'static>, Output<'static>>) {
418    let mut ticker = Ticker::every(Duration::from_millis(1100));
419    loop {
420        ticker.next().await;
421        if watchdog.send_pulse().await.is_err() {
422            error!("watchdog pulse failed");
423        }
424    }
